```python
import aihub_runtime as aihub
from datetime import datetime, timedelta

# Calculate last week's Monday–Sunday
today = datetime.now().date()
days_since_monday = today.weekday()  # 0=Mon, 6=Sun
last_sunday = today - timedelta(days=days_since_monday + 1)
last_monday = last_sunday - timedelta(days=6)

# Query sales by store for last week
rows = aihub.query("EDW (Postgres)", """
  SELECT store, SUM(quantity) as total_quantity, COUNT(*) as transaction_count
  FROM public.sales
  WHERE data_date BETWEEN ? AND ?
  GROUP BY store
  ORDER BY total_quantity DESC
""", [last_monday, last_sunday])

# Log the data
aihub.log(f"Sales for {last_monday} to {last_sunday}: {len(rows)} stores")

# Email the report (spreadsheet export happens separately)
aihub.send_email(
    to=["alex@company.com"],  # ← needs your actual email
    subject=f"Weekly Store Sales Report: {last_monday} to {last_sunday}",
    body=f"Last week's sales by store:\n\n" + 
         "\n".join([f"{r['store']}: {r['total_quantity']} units ({r['transaction_count']} transactions)" for r in rows])
)
```
